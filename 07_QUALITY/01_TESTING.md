        "hostname": "test-pc",
        "bootstrap_token": bootstrap_token,
    })
    assert response.status_code == 200
    assert "client_secret" in response.json()

def test_record_idempotency(client, registered_computer):
    record = {...}
    r1 = client.post("/api/v1/records/batch", json={"records": [record]})
    r2 = client.post("/api/v1/records/batch", json={"records": [record]})
    assert r1.status_code == 200
    assert r2.status_code == 200
    assert r2.json()["rejected_uuids"] == [record["record_uid"]]
