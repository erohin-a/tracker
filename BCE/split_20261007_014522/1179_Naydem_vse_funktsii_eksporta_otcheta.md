funcs = [
    "_report_to_table",
    "_render_csv",
    "_render_xlsx",
    "_render_pdf",
]

for name in funcs:
    m = re.search(rf"def {re.escape(name)}\(.*?(?=\ndef |\n# ============|\Z)",
                  content, re.DOTALL)
    print("=" * 70)
    print(f"=== {name} ===")
    print("=" * 70)
    if m:
        body = m.group(0)
        # Печатаем первые 60 строк, чтобы не раздувать вывод
        lines = body.split("\n")
        for i, line in enumerate(lines[:60], 1):
            print(f"{i:3} {line}")
        if len(lines) > 60:
            print(f"... (ещё {len(lines)-60} строк)")
    else:
        print("НЕ НАЙДЕНА")
    print()
